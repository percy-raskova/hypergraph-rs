```javascript
export const meta = {
  name: 'a2-preflight-pass3',
  description: 'Third adversarial pre-flight of the repaired A2 plan',
  phases: [{ title: 'Find', detail: '5 lenses x 2 scopes, Opus' },
           { title: 'Verify', detail: 'one refute-by-default skeptic per finding' }],
}
const PLAN = 'docs/superpowers/plans/2026-07-31-xgi-compat-a2-cross-register.md'
const LENSES = [
  { key: 'runnable', q: 'Would every code block RUN as written — syntax, paths that exist, commands executable in this repo?' },
  { key: 'facts', q: 'Is every cited fact (file paths, line numbers, counts, quoted code, quoted current-state strings) TRUE of the current tree? Verify each against the tree; trust nothing.' },
  { key: 'consistency', q: 'Does any step contradict the plan\'s own Global Constraints section or another step?' },
  { key: 'property', q: 'Does each test and mutation pin the PROPERTY it claims, or only a proxy for it?' },
  { key: 'compose', q: 'Do the five tasks compose in order — does each step\'s precondition hold given every earlier step (and only earlier steps) ran?' },
]
const SCOPES = [
  { name: 'plan', instr: 'over the ENTIRE plan' },
  { name: 'task5', instr: 'over Task 5 ONLY (the tools/-package conversion) — read every line of it; two prior passes failed to converge on this task' },
]
const FINDINGS = { type: 'object', required: ['findings'], properties: { findings: { type: 'array', items: {
  type: 'object', required: ['task', 'quote', 'claim', 'severity'], properties: {
    task: { type: 'string' }, quote: { type: 'string' }, claim: { type: 'string' },
    severity: { enum: ['critical', 'major', 'minor'] } } } } } }
const VERDICT = { type: 'object', required: ['refuted', 'reasoning'], properties: {
  refuted: { type: 'boolean' }, reasoning: { type: 'string' } } }

phase('Find')
const raw = (await parallel(SCOPES.flatMap(s => LENSES.map(l => () =>
  agent(
    `Adversarial pre-flight (pass 3, ${args.date}) of ${PLAN}, lens=${l.key}, ${s.instr}. ` +
    l.q + ' Two halves: (a) verify the fixes listed in the plan\'s RESOLUTION record are correct as applied; ' +
    '(b) find defects the repair introduced or earlier passes missed. Every finding needs task, exact quote, ' +
    'the defect claim, and severity. Work in the repo; read files; run read-only commands. Return findings only.',
    { label: `find:${s.name}:${l.key}`, phase: 'Find', model: 'opus', schema: FINDINGS }
  ).then(r => (r ? r.findings.map(f => ({ ...f, scope: s.name, lens: l.key })) : []))
)))).filter(Boolean).flat()

const key = f => `${f.task}::${f.quote.slice(0, 60).toLowerCase()}`
const seen = new Set()
const deduped = raw.filter(f => !seen.has(key(f)) && seen.add(key(f)))
log(`${raw.length} raw findings, ${deduped.length} after dedup`)

phase('Verify')
const CAP = 8
const byGroup = {}
for (const f of deduped) {
  const g = `${f.scope}:${f.lens}`
  if (!byGroup[g]) byGroup[g] = []
  byGroup[g].push(f)
}
const toVerify = [], dropped = []
for (const [g, fs] of Object.entries(byGroup)) {
  const ranked = fs.slice().sort((a, b) =>
    ['critical', 'major', 'minor'].indexOf(a.severity) - ['critical', 'major', 'minor'].indexOf(b.severity))
  toVerify.push(...ranked.slice(0, CAP))
  if (ranked.length > CAP) { dropped.push(...ranked.slice(CAP)); log(`DROPPED ${ranked.length - CAP} from ${g} (cap ${CAP})`) }
}
const verified = await parallel(toVerify.map(f => () =>
  agent(
    `Skeptic. REFUTE this pre-flight finding against ${PLAN} and the tree; default to refuted=true if uncertain. ` +
    `Task: ${f.task}. Quote: ${f.quote}. Claim: ${f.claim}. Check the quote exists, the claim is true of it, and the defect is real.`,
    { label: `verify:${f.lens}:${f.task}`, phase: 'Verify', model: 'opus', schema: VERDICT }
  ).then(v => ({ ...f, refuted: v ? v.refuted : true, reasoning: v ? v.reasoning : 'skeptic died; treat as unverified' }))
))
const confirmed = verified.filter(Boolean).filter(f => !f.refuted)
const perTask = {}
for (const f of confirmed) perTask[f.task] = (perTask[f.task] || 0) + 1
log(`${confirmed.length} confirmed; per-task ${JSON.stringify(perTask)}; ${dropped.length} dropped by cap`)
return { confirmed, perTask, droppedCount: dropped.length, dropped }
```

const confirmed = all.filter(f => f.confirmed)
log(`${all.length} findings, ${confirmed.length} confirmed by majority`)
